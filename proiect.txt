int victorie()
{
    if(moves[8][1]==2 or moves[8][3]==2 or moves[8][5]==2 or moves[8][7]==2)
        return 2;

    if(moves[1][8]==2 and moves[2][7]==3) return 3;
    if(moves[1][6]==2 and moves[2][7]==3 and moves[2][5]==3) return 3;
    if(moves[1][4]==2 and moves[2][3]==3 and moves[2][5]==3) return 3;
    if(moves[1][2]==2 and moves[2][1]==3 and moves[2][3]==3) return 3;

    if(moves[2][1]==2 and moves[1][2]==3 and moves[3][2]==3) return 3;
    if(moves[4][1]==2 and moves[3][2]==3 and moves[5][2]==3) return 3;
    if(moves[6][1]==2 and moves[5][2]==3 and moves[7][2]==3) return 3;


    if(moves[3][8]==2 and moves[2][7]==3 and moves[4][7]==3) return 3;
    if(moves[5][8]==2 and moves[4][7]==3 and moves[6][7]==3) return 3;
    if(moves[7][8]==2 and moves[6][7]==3 and moves[8][7]==3) return 3;

    for(int h=2; h<=7; h++)
        for(int f=2; f<=7; f++)
            if(moves[h][f]==2)
                if(moves[h-1][f-1]==3 and moves[h-1][f+1]==3 and moves[h+1][f-1]==3 and moves[h+1][f+1]==3)
                    return 3;

    return 0;
}

void deseneazaPiese(int tabla [8][8])
{
    for (int i=0; i<8; i++)
    {
        for (int j=0; j<8; j++)
        {
            if(tabla [i][j]==1)
            {
                deseneazaPiesa(225+j*50, 225+i*50, 14, pc, pc);
            }
            else if (tabla [i][j]==2)
            {
                deseneazaPiesa(225+j*50, 225+i*50, 14, pv, pv);
            }
        }
    }
}

void deseneazaPiesa(int x, int y, int radius, int border_color, int fill_color)
{
    setfillstyle(SOLID_FILL, fill_color);
    setcolor(border_color);
    circle(x, y, radius);
    floodfill(x, y, border_color);
}
